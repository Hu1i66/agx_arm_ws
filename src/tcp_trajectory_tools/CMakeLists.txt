cmake_minimum_required(VERSION 3.16)
project(tcp_trajectory_tools)
find_package(ament_cmake REQUIRED)
find_package(moveit_core REQUIRED)
find_package(nlohmann_json REQUIRED)
add_executable(tcp_time_parameterize src/time_parameterize.cpp)
target_compile_features(tcp_time_parameterize PRIVATE cxx_std_17)
ament_target_dependencies(tcp_time_parameterize moveit_core)
target_link_libraries(tcp_time_parameterize nlohmann_json::nlohmann_json)
install(TARGETS tcp_time_parameterize DESTINATION lib/${PROJECT_NAME})
ament_package()
