cmake_minimum_required(VERSION 3.16)
project(piper_workbench)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(ament_cmake REQUIRED)
find_package(Qt5 REQUIRED COMPONENTS Widgets Network Concurrent)
find_package(rclcpp REQUIRED)
find_package(rviz_common REQUIRED)
find_package(rviz_rendering REQUIRED)
find_package(rviz_default_plugins REQUIRED)
find_package(moveit_ros_visualization REQUIRED)
find_package(moveit_ros_planning_interface REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_library(MOVEIT_RVIZ_CORE moveit_motion_planning_rviz_plugin_core HINTS /opt/ros/humble/lib REQUIRED)
file(GLOB WORKBENCH_SOURCES CONFIGURE_DEPENDS src/*.cpp src/pages/*.cpp)
file(GLOB WORKBENCH_HEADERS CONFIGURE_DEPENDS include/piper_workbench/*.hpp)
add_executable(piper_workbench ${WORKBENCH_SOURCES} ${WORKBENCH_HEADERS})
target_include_directories(piper_workbench PRIVATE include)
target_link_libraries(piper_workbench Qt5::Widgets Qt5::Network Qt5::Concurrent ${MOVEIT_RVIZ_CORE})
ament_target_dependencies(piper_workbench rclcpp rviz_common rviz_rendering rviz_default_plugins moveit_ros_visualization moveit_ros_planning_interface sensor_msgs geometry_msgs std_msgs tf2_ros)
install(TARGETS piper_workbench DESTINATION lib/piper_workbench)
install(DIRECTORY resources launch DESTINATION share/piper_workbench)
ament_package()
if(BUILD_TESTING)
  add_executable(workbench_state_regressions tests/state_regressions.cpp)
  target_include_directories(workbench_state_regressions PRIVATE include)
  target_link_libraries(workbench_state_regressions Qt5::Core)
  add_executable(workbench_backend_lifecycle tests/backend_lifecycle.cpp src/backend_client.cpp include/piper_workbench/backend_client.hpp)
  target_include_directories(workbench_backend_lifecycle PRIVATE include)
  target_link_libraries(workbench_backend_lifecycle Qt5::Core Qt5::Network)
endif()
